# ============================================================
# 三、编译 CUDA 自定义算子 (Deformable Attention)
# ============================================================
# 这是最关键的一步。DINO 依赖自定义 CUDA kernel 实现可变形注意力。

# 3.1 先确认 CUDA toolkit 可用
nvcc --version
echo $CUDA_HOME  # 如果没有设置，找到 nvcc 所在目录
# 例如: export CUDA_HOME=/usr/local/cuda-11.8

# 3.2 修复源码兼容性问题 (PyTorch 2.x)
# 项目代码中有两处需要修复才能在 PyTorch 2.x 下编译：

# 修复 1: THC → ATen 头文件
sed -i 's|<THC/THCAtomics.cuh>|<ATen/cuda/Atomic.cuh>|' \
  models/dino/ops/src/cuda/ms_deform_im2col_cuda.cuh

# 修复 2: AT_ASSERTM → TORCH_CHECK
sed -i 's/AT_ASSERTM/TORCH_CHECK/g' \
  models/dino/ops/src/cuda/ms_deform_attn_cuda.cu

# 3.3 编译
cd models/dino/ops

# V100 = SM 7.0
export TORCH_CUDA_ARCH_LIST="7.0"

# CUDA_HOME: conda 安装的 cuda-toolkit 在 conda 环境根目录下
export CUDA_HOME=$CONDA_PREFIX
# 如果是系统级安装: export CUDA_HOME=/usr/local/cuda-12.3

# 编译 + 安装（nvcc 12.3 + PyTorch 2.5.1+cu121 版本完全匹配）
python setup.py build develop

# 3.4 验证编译
cd ../../..
python -c "
import MultiScaleDeformableAttention as MSDA
print('CUDA 算子编译成功:', dir(MSDA))
"

# 3.5 如果编译失败：检查清单
# - nvcc 版本和 PyTorch CUDA 版本是否匹配 (都必须是指定 CUDA 版本)
# - gcc/g++ 版本 (推荐 7-11, 太新可能报错)
# - CUDA_HOME 是否设置正确

# ============================================================
# 六、训练命令
# ============================================================

# 6.1 单 GPU 训练 (调试用)
python main.py \
  --config_file config/DINO/DINO_4scale_multimodal.py \
  --coco_path /path/to/your/tri-modal/dataset/ \
  --output_dir ./output/test_run \
  --batch_size 2

# 6.2 4×V100 分布式训练 (生产用)
# 每个 GPU batch_size=1, 总 batch_size=4
python -m torch.distributed.launch --nproc_per_node=4 main.py \
  --config_file config/DINO/DINO_4scale_multimodal.py \
  --coco_path /path/to/your/tri-modal/dataset/ \
  --output_dir ./output/multimodal_full \
  --batch_size 1 \
  --num_workers 4

# 6.3 消融实验 —— 纯 RGB 基线
python -m torch.distributed.launch --nproc_per_node=4 main.py \
  --config_file config/DINO/DINO_4scale_multimodal.py \
  --coco_path /path/to/your/tri-modal/dataset/ \
  --output_dir ./output/ablation_rgb_only \
  --batch_size 1 \
  --options fusion_modality=rgb_only

# 6.4 如果要从 COCO 预训练权重微调
python main.py \
  --config_file config/DINO/DINO_4scale_multimodal.py \
  --coco_path /path/to/your/tri-modal/dataset/ \
  --output_dir ./output/finetune \
  --pretrain_model_path /path/to/DINO_COCO_pretrain.pth \
  --finetune_ignore label_enc.weight class_embed \
  --batch_size 2


# ============================================================
# 七、常见问题
# ============================================================

# Q1: CUDA out of memory
# → --batch_size 1, 每卡 batch=1
# → 检查是否有其他进程占用 GPU

# Q2: nvcc 报错 "unsupported gcc version"
# → conda install -c conda-forge gxx_linux-64==11.2.0
# → 或降低 gcc 版本

# Q3: pycocotools 编译失败
# → pip install cython numpy 先装好再试
# → 或者直接从 conda 安装: conda install -c conda-forge pycocotools

# Q4: CUDA 算子编译失败 "THC/THCAtomics.cuh: No such file"
# → 确认已执行步骤 3.2 的 sed 修复命令

# Q5: import MultiScaleDeformableAttention 失败
# → 确认已执行 setup.py develop (步骤 3.3)
# → export LD_LIBRARY_PATH=/usr/local/cuda-12.3/lib64:$LD_LIBRARY_PATH
